  function initBackgroundCanvas() {
    if (!canvas) {
      return;
    }

    const ctx = canvas.getContext("2d", { alpha: true });
    if (!ctx) {
      return;
    }

    let width = 0;
    let height = 0;
    let sweepAngle = 0;
    const nodes = [];
    const nodeCount = 42;

    function resize() {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      width = window.innerWidth;
      height = window.innerHeight;
      canvas.width = Math.floor(width * dpr);
      canvas.height = Math.floor(height * dpr);
      canvas.style.width = width + "px";
      canvas.style.height = height + "px";
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

      nodes.length = 0;
      for (let i = 0; i < nodeCount; i += 1) {
        nodes.push({
          x: Math.random() * width,
          y: Math.random() * height,
          r: 0.7 + Math.random() * 1.6,
          phase: Math.random() * Math.PI * 2,
          speed: 0.0004 + Math.random() * 0.001
        });
      }
    }

    function drawFrame(ts) {
      ctx.clearRect(0, 0, width, height);

      ctx.strokeStyle = "rgba(0, 229, 255, 0.07)";
      ctx.lineWidth = 1;
      const gap = 70;
      for (let x = 0; x < width; x += gap) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, height);
        ctx.stroke();
      }
      for (let y = 0; y < height; y += gap) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(width, y);
        ctx.stroke();
      }

      for (let i = 0; i < nodes.length; i += 1) {
        const node = nodes[i];
        const alpha = 0.22 + Math.sin(ts * node.speed + node.phase) * 0.16;
        ctx.fillStyle = "rgba(0,255,157," + Math.max(0.08, alpha).toFixed(3) + ")";
        ctx.beginPath();
        ctx.arc(node.x, node.y, node.r, 0, Math.PI * 2);
        ctx.fill();

        if (i % 5 === 0 && nodes[i + 1]) {
          ctx.strokeStyle = "rgba(0, 229, 255, 0.05)";
          ctx.beginPath();
          ctx.moveTo(node.x, node.y);
          ctx.lineTo(nodes[i + 1].x, nodes[i + 1].y);
          ctx.stroke();
        }
      }

      const cx = width * 0.8;
      const cy = height * 0.22;
      const radius = Math.min(width, height) * 0.22;
      sweepAngle += 0.01;

      ctx.strokeStyle = "rgba(0,255,157,0.12)";
      ctx.beginPath();
      ctx.arc(cx, cy, radius, 0, Math.PI * 2);
      ctx.stroke();

      const ex = cx + Math.cos(sweepAngle) * radius;
      const ey = cy + Math.sin(sweepAngle) * radius;
      const grad = ctx.createLinearGradient(cx, cy, ex, ey);
      grad.addColorStop(0, "rgba(0,255,157,0.02)");
      grad.addColorStop(1, "rgba(0,255,157,0.22)");
      ctx.strokeStyle = grad;
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.lineTo(ex, ey);
      ctx.stroke();

      requestAnimationFrame(drawFrame);
    }

    window.addEventListener("resize", resize);
    resize();
    requestAnimationFrame(drawFrame);
  }

  /**

    // Format plain text into styled HTML paragraphs and bullet points
    function formatText(text) {
        if (!text) return '';
        let lines = text.split('\n');
        let html = '';
        let inList = false;

        lines.forEach(line => {
            let trimmed = line.trim();
            if (trimmed.startsWith('•') || trimmed.startsWith('-')) {
                if (!inList) { html += '<ul>'; inList = true; }
                html += `<li>${trimmed.substring(1).trim()}</li>`;
            } else {
                if (inList) { html += '</ul>'; inList = false; }
                if (trimmed.length > 0) {
                    html += `<p>${trimmed}</p>`;
                }
            }
        });
        if (inList) html += '</ul>';
        return html;
    }

    function savePersonalOC() {
        const ocData = {
            id: 'oc_' + Date.now(),
            owner: currentCallsign,
            name: document.getElementById('oc-name').value,
            race: document.getElementById('oc-race').value,
            power: document.getElementById('oc-power').value,

                hp: document.getElementById('oc-hp').value,

                stm: document.getElementById('oc-stm').value,
                str: document.getElementById('oc-str').value
            },
            skills: document.getElementById('oc-skills').value,
            timestamp: new Date().toISOString()
        };

        const allOcs = JSON.parse(localStorage.getItem('pantheon_user_oc') || '[]');
        allOcs.push(ocData);
        localStorage.setItem('pantheon_user_oc', JSON.stringify(allOcs));

        const status = document.getElementById('oc-status');
        status.style.display = 'block';
        setTimeout(() => { status.style.display = 'none'; }, 2500);

        renderPublicShowcaseCard();
        renderUserAssets();
    }

    function uploadMediaAsset(e) {
        e.preventDefault();
        const title = document.getElementById('media-title').value;
        const privacy = document.getElementById('media-privacy').value;
        const fileInput = document.getElementById('media-file');

        if (!fileInput.files || !fileInput.files[0]) return;

        const reader = new FileReader();
        reader.onload = function(evt) {
            const mediaData = {
                id: 'media_' + Date.now(),
                owner: currentCallsign,
                title: title,
                privacy: privacy,
                data: evt.target.result,
                timestamp: new Date().toISOString()
            };

            const allMedia = JSON.parse(localStorage.getItem('pantheon_media') || '[]');
            allMedia.push(mediaData);
            localStorage.setItem('pantheon_media', JSON.stringify(allMedia));

            const status = document.getElementById('media-status');
            status.style.display = 'block';
            setTimeout(() => { status.style.display = 'none'; }, 2500);

            document.getElementById('media-title').value = '';
            fileInput.value = '';
            renderUserAssets();
        };

        reader.readAsDataURL(fileInput.files[0]);
    }

    function renderUserAssets() {
        // Render Personal OCs
        const ocBox = document.getElementById('user-oc-list');
        const allOcs = JSON.parse(localStorage.getItem('pantheon_user_oc') || '[]');
        const myOcs = allOcs.filter(o => o.owner.toLowerCase() === currentCallsign.toLowerCase());

        if (myOcs.length === 0) {
            ocBox.innerHTML = '<div style="color:#607080; font-size:0.75rem;">No personal OCs created yet.</div>';
        } else {
            ocBox.innerHTML = myOcs.map(oc => `
                <div class="asset-box">
                    <div class="asset-title">${oc.name.toUpperCase()}</div>
                    <div style="font-size:0.7rem; color:var(--db-ki); margin-top:2px;">${oc.race} // POWER: ${oc.power}</div>
                    <div style="font-size:0.7rem; color:#8a9ba8; margin-top:4px;">Skills: ${oc.skills}</div>
                    <div class="asset-meta">
                        <span>Created: ${new Date(oc.timestamp).toLocaleDateString()}</span>
                        <button class="btn-del" onclick="deleteOc('${oc.id}')">[DELETE]</button>
                    </div>
                </div>
            `).join('');
        }

        // Render Uploaded Media
        const mediaBox = document.getElementById('user-media-list');
        const allMedia = JSON.parse(localStorage.getItem('pantheon_media') || '[]');
        const myMedia = allMedia.filter(m => m.owner.toLowerCase() === currentCallsign.toLowerCase());

        if (myMedia.length === 0) {
            mediaBox.innerHTML = '<div style="color:#607080; font-size:0.75rem;">No media assets transmitted yet.</div>';
        } else {
            mediaBox.innerHTML = myMedia.map(m => `
                <div class="asset-box">
                    <img src="${m.data}" alt="${m.title}">
                    <div class="asset-title">${m.title}</div>
                    <div class="asset-meta">
                        <span style="color:${m.privacy === 'public' ? 'var(--db-ki)' : 'var(--db-gold)'};">[${m.privacy.toUpperCase()}]</span>
                        <button class="btn-del" onclick="deleteMedia('${m.id}')">[DELETE]</button>
                    </div>
                </div>
            `).join('');
        }
    }

    function deleteOc(id) {
        let allOcs = JSON.parse(localStorage.getItem('pantheon_user_oc') || '[]');
        allOcs = allOcs.filter(o => o.id !== id);
        localStorage.setItem('pantheon_user_oc', JSON.stringify(allOcs));
        renderPublicShowcaseCard();
        renderUserAssets();
    }

    function deleteMedia(id) {

        allMedia = allMedia.filter(m => m.id !== id);
        localStorage.setItem('pantheon_media', JSON.stringify(allMedia));
        renderUserAssets();
    }

                        (localStorage.getItem('pantheon_director_authenticated') === 'true');

        adminLinks.forEach(link => {

        });
